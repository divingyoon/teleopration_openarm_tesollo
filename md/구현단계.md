- Phase 2: Action Representation 수정
    
    목적: joint-space delta → task-space 6D delta-pose
    
    현재 right_palm_delta = _delta(right_js[:, :6]) (관절 차분) → 변경 필요:
    
    ```python
      def joint_to_taskspace_delta(q_curr: np.ndarray, q_next: np.ndarray) -> np.ndarray:                
          """                                                                                            
          두 관절 상태로부터 palm frame EEF delta pose 계산
          Returns: [dx, dy, dz, drx, dry, drz] (axis-angle representation)                               
          """                                                                                            
          T_curr = fk_right_palm(q_curr)  # 4x4                                                          
          T_next = fk_right_palm(q_next)  # 4x4                                                          
                                                                
          # Relative transform: T_delta = T_curr^{-1} @ T_next (palm frame)                              
          T_delta = np.linalg.inv(T_curr) @ T_next              
                                                                                                         
          translation = T_delta[:3, 3]           # [dx, dy, dz]                                          
          rot = Rotation.from_matrix(T_delta[:3, :3])                                                    
          axis_angle = rot.as_rotvec()           # [drx, dry, drz]                                       
                                                                                                         
          return np.concatenate([translation, axis_angle])  # [6]
    ```
    
    이렇게 변환된 6D action이 PourMimicEnv.action_to_target_eef_pose()와 정렬됨.

    ```python
      def compute_datagen_info(joint_traj_right, joint_traj_left, cup_poses):
          T = len(joint_traj_right)                                                                      
                                                                
          # EEF pose sequences                                                                           
          eef_right = np.stack([fk_right_palm(q) for q in joint_traj_right])  # [T, 4, 4]
          eef_left  = np.stack([fk_left_palm(q)  for q in joint_traj_left])   # [T, 4, 4]                
                                                                
          # Target EEF pose: next timestep's pose (t+1 at time t, last=last)                             
          target_right = np.concatenate([eef_right[1:], eef_right[-1:]], axis=0)
          target_left  = np.concatenate([eef_left[1:],  eef_left[-1:]],  axis=0)                         
                                                                                                         
          # Object poses: cup0=source_cup, cup1=target_cup                                               
          src_cup_poses  = cup_poses["cup0"]   # [T, 4, 4]                                               
          tgt_cup_poses  = cup_poses["cup1"]   # [T, 4, 4]                                               
                                                                                                         
          return {                                                                                       
              "eef_pose/right":            eef_right,                                                    
              "eef_pose/left":             eef_left,            
              "target_eef_pose/right":     target_right,                                                 
              "target_eef_pose/left":      target_left,
              "object_pose/source_cup":    src_cup_poses,                                                
              "object_pose/target_cup":    tgt_cup_poses,                                                
          }
    
    ```

    ```python
      def compute_subtask_signals(eef_right, cup0_pose, cup1_pose, hand_curl):
          T = len(eef_right)                                                                             
          grasp_done = np.zeros((T, 1), dtype=np.float32)       
          lift_done  = np.zeros((T, 1), dtype=np.float32)                                                
          align_done = np.zeros((T, 1), dtype=np.float32)       
          pour_done  = np.zeros((T, 1), dtype=np.float32)                                                
                                                                
          # Heuristic 예시:                                                                              
          # grasp_done: hand curl avg > threshold (컵을 잡은 상태)
          curl_avg = hand_curl.mean(axis=1)                                                              
          grasp_t  = np.argmax(curl_avg > GRASP_CURL_THRESHOLD) 
                                                                                                         
          # lift_done: EEF z > table_height + lift_margin                                                
          lift_t   = np.argmax(eef_right[:, 2, 3] > LIFT_HEIGHT_THRESHOLD)                               
                                                                                                         
          # align_done: EEF xy 위치가 target_cup xy에 근접                                               
          dist_xy  = np.linalg.norm(eef_right[:, :2, 3] - cup1_pose[:, :2, 3], axis=1)                   
          align_t  = np.argmax(dist_xy < ALIGN_DIST_THRESHOLD)                                           
                                                                
          # pour_done: 마지막 프레임                                                                     
          pour_t   = T - 1                                      
                                                                                                         
          # 0→1 transition 설정                                                                          
          if grasp_t > 0: grasp_done[grasp_t:] = 1.0
          if lift_t > 0:  lift_done[lift_t:]   = 1.0                                                     
          if align_t > 0: align_done[align_t:] = 1.0                                                     
          pour_done[pour_t:]                    = 1.0                                                    
                                                                                                         
          return grasp_done, lift_done, align_done, pour_done   
    ```
